Разбери пример «Надёжный ввод» по строкам:

def ask(question, options):
    answer = input(question)
    while answer not in options:
        print("Выбери из:", ", ".join(options))
        answer = input(question)
    return answer

choice = ask("Налево или направо? ", ["налево", "направо"])
print("Ты пошёл", choice)
