Разбери пример «Шифр Цезаря» по строкам:

ABC = "абвгдеёжзийклмнопрстуфхцчшщъыьэюя"

def encrypt(text, key):
    result = ""
    for ch in text.lower():
        if ch in ABC:
            result += ABC[(ABC.index(ch) + key) % len(ABC)]
        else:
            result += ch
    return result

secret = encrypt("встречаемся в полночь", 3)
print(secret)
for key in range(1, 6):
    print(key, encrypt(secret, -key))
