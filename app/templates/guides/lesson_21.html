2. Ключевые темы

  • `Rect` и `colliderect`
  • Текст на экране
  • Жизни и Game Over

Разбери пример «Ловим яблоки» по строкам:

import pygame, random
pygame.init()
screen = pygame.display.set_mode((600, 600))
clock = pygame.time.Clock()
font = pygame.font.SysFont(None, 40)
basket = pygame.Rect(260, 560, 80, 20)
apple = pygame.Rect(random.randint(0, 580), 0, 20, 20)
score, lives, speed = 0, 3, 4
while lives > 0:
    for e in pygame.event.get():
        if e.type == pygame.QUIT:
            lives = 0
    keys = pygame.key.get_pressed()
    basket.x += (keys[pygame.K_RIGHT] - keys[pygame.K_LEFT]) * 8
    apple.y += speed
    if apple.colliderect(basket):
        score += 1; speed += 0.3
        apple.topleft = (random.randint(0, 580), 0)
    elif apple.y > 600:
        lives -= 1
        apple.topleft = (random.randint(0, 580), 0)
    screen.fill((135, 206, 235))
    pygame.draw.rect(screen, (139, 69, 19), basket)
    pygame.draw.circle(screen, (220, 20, 60), apple.center, 10)
    screen.blit(font.render(f"Очки: {score}  Жизни: {lives}", True, (0, 0, 0)), (10, 10))
    pygame.display.flip()
    clock.tick(60)
pygame.quit()
print("Игра окончена! Очки:", score)
