Разбери пример «Частицы (логика)» по строкам:

import random

particles = []

def explode(x, y):
    for _ in range(20):
        particles.append({"x": x, "y": y,
                          "dx": random.uniform(-3, 3), "dy": random.uniform(-3, 3),
                          "life": 30})

def update():
    for p in particles:
        p["x"] += p["dx"]; p["y"] += p["dy"]; p["life"] -= 1
    particles[:] = [p for p in particles if p["life"] > 0]

explode(100, 100)
for frame in range(31):
    update()
print("Осталось частиц:", len(particles))
