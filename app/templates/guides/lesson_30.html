Разбери пример «Ближайший сосед» по строкам:

import math
# (сладость 0..10, хруст 0..10) -> что это
examples = [((9, 3), "фрукт"), ((8, 6), "фрукт"), ((2, 9), "овощ"), ((1, 7), "овощ")]

def guess(sweet, crunch):
    best = min(examples, key=lambda e: math.dist(e[0], (sweet, crunch)))
    return best[1]

print("Морковка (3, 9):", guess(3, 9))
print("Банан (8, 1):", guess(8, 1))
